```
   programa.Id = PromoProgramaId.CreateNew()

9. var creado = await _promoProgramaService.GetByIdAsync(programaId.Value, ct)  // recargar para el DTO
   var dto = _mapper.Map<PromoProgramaCreatedResultDto>(creado)
```

**Flujo completo del Handle:**

```
1. await _validator.ValidateAsync(request, ct)
   -> Si invalido: LogWarning + return ServiceResponse con errores de validacion

2. Si string.IsNullOrEmpty(request.UserId):
   -> return ValidateExtensions.BadRequestServiceResponse<PromoProgramaUpdatedResultDto>(
          "Token invalido", ServiceResponseMessageType.Auth_Unauthorized)

3. var artista = await _promoProgramaService.GetArtistaByUserIdAsync(request.UserId, ct)
   -> Si null: return ValidateExtensions.NotFoundServiceResponse<PromoProgramaUpdatedResultDto>(
          "No tienes un perfil de artista", ServiceResponseMessageType.NotFound_Artista)

4. var programa = await _promoProgramaService.GetByIdAsync(request.Id, ct)
   -> Si null: return ValidateExtensions.NotFoundServiceResponse<PromoProgramaUpdatedResultDto>(
          "El programa de promocion no existe", ServiceResponseMessageType.NotFound_PromoPrograma)

5. Si programa.ArtistaId != artista.Id:
   -> return ValidateExtensions.ForbiddenServiceResponse<PromoProgramaUpdatedResultDto>(
          "No tienes permiso para modificar este programa", ServiceResponseMessageType.Auth_Forbidden)

6. Si request.CampaniaCrowdfundingId.HasValue:
   var pertenece = await _promoProgramaService.VerificarCampaniaPertenece(artista.Id, request.CampaniaCrowdfundingId.Value, ct)
   -> Si false: return ValidateExtensions.ForbiddenServiceResponse<PromoProgramaUpdatedResultDto>(
          "La campana no pertenece a tu perfil de artista", ServiceResponseMessageType.Auth_Forbidden)

7. Actualizar campos del programa (logica de negocio en el handler, NO en el service):
   programa.Titulo = request.Titulo
   programa.Descripcion = request.Descripcion
   programa.TipoPromoId = request.TipoPromoId
   programa.CampaniaCrowdfundingId = request.CampaniaCrowdfundingId

   programa.CodigoTrackingBase = request.CodigoTrackingBase

   programa.ImporteComisionPorcentaje = request.ImporteComisionPorcentaje
   programa.ImporteComisionFija = request.ImporteComisionFija
   programa.FechaInicio = request.FechaInicio.HasValue ? request.FechaInicio.Value.ToDateTime(TimeOnly.MinValue) : null
   programa.FechaFin = request.FechaFin.HasValue ? request.FechaFin.Value.ToDateTime(TimeOnly.MinValue) : null
   programa.FechaActualizacion = DateTime.UtcNow

8. Clasificar tareas recibidas:
   var tareasNuevas = new List<PromoTarea>()
   var tareasActualizar = new List<PromoTarea>()
   var tareasDesactivar = new List<Guid>()   // IDs de tareas a desactivar (EsActivo == false y Id presente)

   foreach (var tareaItem in request.Tareas):
       Si tareaItem.Id == null (nueva tarea):
           var nuevaTarea = _mapper.Map<PromoTarea>(tareaItem)
           nuevaTarea.Id = PromoTareaId.CreateNew()
           nuevaTarea.ProgramaId = programa.Id
           nuevaTarea.EsActivo = true
           nuevaTarea.Orden = calcular orden (max orden actual + posicion)
           nuevaTarea.FechaCreacion = DateTime.UtcNow
           tareasNuevas.Add(nuevaTarea)
       Si tareaItem.Id presente && tareaItem.EsActivo == false:
           tareasDesactivar.Add(tareaItem.Id.Value)
           (la validacion de completados ya se hizo en el validator)
       Si tareaItem.Id presente && tareaItem.EsActivo == true:

           tareaExistente.Id = PromoTareaId.From(tareaItem.Id.Value)
           tareaExistente.ProgramaId = programa.Id
           tareasActualizar.Add(tareaExistente)

10. var dto = _mapper.Map<PromoProgramaUpdatedResultDto>(programa)

11. _logger.LogInformation("PromoPrograma {ProgramaId} updated by ArtistaId {ArtistaId}",

12. return new ServiceResponse<PromoProgramaUpdatedResultDto>
    {
        Data = dto,
        Messages = new List<ServiceResponseMessage>
        {
            new() { Message = "Programa de promocion actualizado", HttpStatusCode = System.Net.HttpStatusCode.OK }
        }
    }
```

## 16. Decisiones de Diseno

### 16.1 Clases internas para items de tarea

Los commands `CreatePromoProgramaCommand` y `UpdatePromoProgramaCommand` definen las clases `CreatePromoTareaItem` y `UpdatePromoTareaItem` como clases internas (o anidadas) en el mismo archivo del command. Esto sigue el patron establecido en el proyecto donde el command file contiene toda la estructura que necesita el handler, y evita crear archivos extra solo para tipos auxiliares.

Las referencias en los AutoMapper profiles usan el nombre cualificado: `CreatePromoProgramaCommand.CreatePromoTareaItem`.

### 16.2 GetMisProgramas sin validator

La query `GetMisProgramasQuery` no tiene validator porque:
- Los parametros de paginacion (`page`, `pageSize`) se clampen en el controller antes de enviar la query
- `EsActivo` es un `bool?` que no requiere validacion de formato
- El `UserId` se verifica directamente en el handler (patron identico a `GetPromotorMeQuery`)

### 16.3 GetPromoProgramaByIdQuery sin validator

Misma razon: el `id` se tipea como `Guid` en el route constraint `{id:guid}`, lo que garantiza formato valido antes de llegar al handler.

### 16.4 DesactivarPromoProgramaCommand sin IMapper

El DesactivarHandler construye el DTO de respuesta directamente (`new PromoProgramaDesactivadoResultDto { ... }`) en lugar de usar AutoMapper. Esto es identico al patron de `DesactivarPromotorCommand` que construye `PromotorDesactivadoResultDto` manualmente. La razon es que el DTO tiene un campo calculado (`TareasDesactivadas`) que se obtiene del retorno del service, haciendo el mapping manual mas simple que configurar AutoMapper para un caso puntual.

### 16.5 Resolucion de nombres de maestras en memoria

Los handlers usan `Dictionary<int, string>` estaticos y privados para resolver los nombres de TipoPromo, TipoEventoPromo, TipoReward y Moneda. Este patron es identico al `TipoPromotorNombres` ya usado en `CreatePromotorCommandHandler` y `GetPromotorMeQueryHandler`. No se hacen llamadas a la base de datos para resolver estos nombres en el MVP.

### 16.6 GetResumenAsync retorna DTO directamente

El metodo `GetResumenAsync` en `IPromoProgramaService` es la unica excepcion a la regla "services retornan entidades". Retorna `PromoProgramaResumenDto` porque el resumen es un calculo agregado (SUM, COUNT) que no corresponde a ninguna entidad del dominio. Es similar a como `PromotorService.GetProgramasActivosCountAsync` retorna un `int` directamente.

### 16.7 Orden de declaracion en controller

El action `GetMisProgramas` (ruta literal `mis-programas`) DEBE declararse antes del action `GetById` (ruta template `{id:guid}`) para evitar que ASP.NET Core intente parsear "mis-programas" como un Guid. Aunque el constraint `:guid` deberia prevenir el conflicto, el orden de declaracion garantiza el comportamiento correcto y sigue las buenas practicas de ASP.NET Core routing.
